180 template <typename Container>
181 static double skaiciuotiVid(const Container& nd) {
182 if (nd.empty()) {
183 throw std::runtime_error("Namu darbu sarasas negali buti tuscias");
184 }
185 double suma = 0.0;
186 for (const auto& elem : nd) {
187 suma += elem;
188 }
189 return suma / nd.size();
190 }

197 template <typename Container>
198 static double skaiciuotiMed(Container nd) {
199 if (nd.empty()) {
200 throw std::runtime_error("Namu darbu sarasas negali buti tuscias");
201 }
202 std::sort(nd.begin(), nd.end());
203 size_t dydis = nd.size();
204 if (dydis % 2 == 0) {
205 return (nd[dydis / 2 - 1] + nd[dydis / 2]) / 2.0;
206 } else {
207 return nd[dydis / 2];
208 }
209 }
